import argparse,json,re,shlex,ssl,sys
from pathlib import Path
sys.path.insert(0,'/mnt/e/Code/PCI/github-export/vmware-esxi8-lab')
from pyVim.connect import SmartConnect,Disconnect
from pyVmomi import vim
from esxi_passthru_capability import EsxiSsh,caveat_field,utc_now
import test_esxi_owner_activation as owner
import test_esxi_dependency_interaction as lab
p=argparse.ArgumentParser();p.add_argument('action',choices=['set','restore']);args=p.parse_args()
run='20261112-esxi8-legacy-bridge-cause-v1';out=Path('/mnt/e/Code/PCI/_artifacts/vmware-directpath-runs')/run;lab.OUT=out
secret=caveat_field('esxi8-auto-codex-credentials','Password');ssh=EsxiSsh('172.31.91.10',secret,retries=1)
si=SmartConnect(host='172.31.91.10',user='root',pwd=secret,sslContext=ssl._create_unverified_context())
try:
    host=owner.inventory(si.RetrieveContent(),vim.HostSystem)[0]
    assert host.hardware.systemInfo.uuid=='6a4e818a-51ce-7d7f-a21f-000c29abaf6d' and host.config.product.build=='24280767'
    assert all(v.runtime.powerState=='poweredOff' for v in host.vm)
    targets=['0000:02:01.0','0000:02:01.1'];mgmt='0000:00:05.0'
    matched={d.id for d in host.hardware.pciDevice if (d.vendorId&65535,d.deviceId&65535)==(0x8086,0x10d3)}
    assert matched==set(targets+[mgmt])
    native=ssh.run('esxcli hardware pci list',timeout=40).stdout
    block=re.search(r'(?ms)^0000:00:05\.0\n.*?(?=^0000:|\Z)',native).group()
    assert all(re.search(x,block) for x in [r'Configured Owner: VMkernel',r'Current Owner: VMkernel',r'FPT Sharable: false'])
    mgmt_state=owner.passthru_state(host,{mgmt});assert mgmt_state and not any(x['capable'] or x['enabled'] or x['active'] for x in mgmt_state)
    original=ssh.run('cat /etc/vmware/passthru.map').stdout
    marker='# Isolated legacy bridge root-cause control '+run
    line='8086 10d3 default false'
    if args.action=='set':
        assert not re.search(r'^\s*8086\s+10d3\s',original,re.M|re.I)
        assert not (out/'control-map-original.txt').exists()
        (out/'control-map-original.txt').write_text(original)
        updated=original.rstrip()+'\n\n'+marker+'\n'+line+'\n'
    else:
        assert marker in original and line in original
        updated=(out/'control-map-original.txt').read_text()
    host.configManager.pciPassthruSystem.UpdatePassthruConfig(config=[vim.host.PciPassthruConfig(id=t,passthruEnabled=False,applyNow=True) for t in targets])
    bank=ssh.run('readlink -f /bootbank').stdout.strip();assert bank.startswith('/vmfs/volumes/')
    ssh.run('printf %s '+shlex.quote(updated)+' > /etc/vmware/passthru.map')
    assert ssh.run('cat /etc/vmware/passthru.map').stdout==updated
    backup=ssh.run('/sbin/auto-backup.sh',timeout=90)
    (out/('control-map-'+args.action+'-backup.txt')).write_text(backup.stdout+backup.stderr)
    record={'at':utc_now(),'action':args.action,'bootTime':str(host.runtime.bootTime),'targets':targets,'managementUnchanged':mgmt_state,'managementNative':block,'bootbank':bank,'rule':line if args.action=='set' else None,'backupExit':backup.returncode}
    lab.save('control-map-'+args.action+'.json',record)
    if args.action=='set':
        if not host.runtime.inMaintenanceMode:
            task,_=lab.task_record(host.EnterMaintenanceMode_Task(timeout=60));assert task['state']=='success'
        r=ssh.run('esxcli system shutdown reboot --delay=10 --reason=legacy-bridge-domain-control',timeout=30)
        print('Persisted dependency policy; normal reboot requested',flush=True)
    else:print('Original policy restored and persisted; targets disabled',flush=True)
finally:Disconnect(si)
