#!/usr/bin/env python3
"""Observe single-member disable requests against the isolated dependency lab."""
import argparse
import json
from pathlib import Path
import re
import ssl
import time

from pyVim.connect import SmartConnect, Disconnect
from pyVmomi import vim
from esxi_passthru_capability import EsxiSsh, caveat_field, utc_now
import test_esxi_owner_activation as owner

OUT = Path("/mnt/e/Code/PCI/_artifacts/vmware-directpath-runs/20261111-esxi8-dependency-disable-v1")
TARGETS = ["0000:02:01.0", "0000:02:01.1"]


def save(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2, default=str) + "\n")


def state(host):
    return [{"id": x.id, "capable": x.passthruCapable, "enabled": x.passthruEnabled,
             "active": x.passthruActive, "dependentDevice": x.dependentDevice}
            for x in host.configManager.pciPassthruSystem.pciPassthruInfo if x.id in TARGETS]


def snapshot(content, host, ssh, label):
    r = ssh.run("esxcli hardware pci list", timeout=45)
    (OUT / (label + "-pci.txt")).write_text(r.stdout)
    native = {}
    for bdf in TARGETS:
        match = re.search(r"(?ms)^" + re.escape(bdf) + r"\n.*?(?=^0000:|\Z)", r.stdout)
        if not match:
            raise RuntimeError("PCI inventory lost the test member " + bdf)
        fields = {}
        for line in match.group().splitlines()[1:]:
            if ": " in line:
                key, _, value = line.strip().partition(":")
                value = value.strip()
                if key in {"Configured Owner", "Current Owner", "Module Name", "Vendor ID", "Device ID", "Dependent Device Address", "Reset Method", "FPT Sharable"}:
                    fields[key] = value
        native[bdf] = fields
    result = {"at": utc_now(), "endpointProduct": str(content.about), "hostProduct": str(host.config.product),
              "hostUuid": host.hardware.systemInfo.uuid, "hostBootTime": str(host.runtime.bootTime),
              "managementServerIp": getattr(host.summary, "managementServerIp", None),
              "hostConfigPci": [{"id": x.id, "capable": x.passthruCapable,
                                  "enabled": x.passthruEnabled, "active": x.passthruActive,
                                  "dependentDevice": x.dependentDevice}
                                 for x in host.config.pciPassthruInfo if x.id in TARGETS],
              "pci": state(host), "native": native}
    save(label + ".json", result)
    return result


def request(content, host, ssh, ids, enabled, label):
    before = snapshot(content, host, ssh, label + "-before")
    result = {"at": utc_now(), "request": {"ids": ids, "passthruEnabled": enabled, "applyNow": True},
              "before": before, "samples": []}
    try:
        host.configManager.pciPassthruSystem.UpdatePassthruConfig(config=[
            vim.host.PciPassthruConfig(id=b, passthruEnabled=enabled, applyNow=True) for b in ids])
        result["callOutcome"] = "success"
    except Exception as error:
        result["callOutcome"] = "fault"
        result["fault"] = owner.fault_snapshot(error)
    started = time.monotonic()
    previous = None
    stable = 0
    while time.monotonic() - started < 30:
        current = state(host)
        result["samples"].append({"at": utc_now(), "pci": current})
        stable = stable + 1 if current == previous else 0
        previous = current
        selected = [x for x in current if x["id"] in ids]
        settled = len(selected) == len(ids) and all(x["enabled"] == enabled and x["active"] == enabled for x in selected)
        if time.monotonic() - started >= 5 and stable >= 2 and (settled or result["callOutcome"] == "fault"):
            break
        time.sleep(1)
    result["after"] = snapshot(content, host, ssh, label + "-after")
    result["changedIds"] = [x["id"] for x in result["after"]["pci"]
                            if x != next(y for y in before["pci"] if y["id"] == x["id"])]
    save(label + ".json", result)
    print(json.dumps({"label": label, "request": result["request"], "outcome": result["callOutcome"],
                      "after": result["after"]["pci"], "native": result["after"]["native"]}), flush=True)
    return result


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("action", choices=["snapshot", "matrix", "prepare-active", "disable-both"])
    parser.add_argument("--endpoint", choices=["esxi", "vcenter"], default="esxi")
    parser.add_argument("--label", default="snapshot")
    args = parser.parse_args()
    if not re.fullmatch(r"[a-z0-9-]+", args.label):
        raise ValueError("Invalid evidence label")
    OUT.mkdir(parents=True, exist_ok=True)
    password = caveat_field("esxi8-auto-codex-credentials", "Password")
    ssh = EsxiSsh("172.31.91.10", password, retries=1)
    if args.endpoint == "esxi":
        server, user, secret = "172.31.91.10", "root", password
    else:
        server = "192.168.91.131"
        user = caveat_field("vcsa803-codex-credentials", "SSO Username")
        secret = caveat_field("vcsa803-codex-credentials", "SSO Password")
    si = SmartConnect(host=server, user=user, pwd=secret, sslContext=ssl._create_unverified_context())
    try:
        content = si.RetrieveContent()
        hosts = [h for h in owner.inventory(content, vim.HostSystem)
                 if h.hardware.systemInfo.uuid == "6a4e818a-51ce-7d7f-a21f-000c29abaf6d"]
        if len(hosts) != 1:
            raise RuntimeError("Test host identity is not unique")
        host = hosts[0]
        if host.config.product.build != "24280767" or host.runtime.connectionState != "connected":
            raise RuntimeError("Expected connected ESXi build 24280767")
        current = state(host)
        if len(current) != 2 or not all(x["capable"] for x in current) or {x["dependentDevice"] for x in current} != {"0000:01:00.0"}:
            raise RuntimeError("Expected shared dependency was not observed")
        for vm in host.vm:
            if vm.runtime.powerState == "poweredOn" and vm.config:
                if any(isinstance(d, vim.vm.device.VirtualPCIPassthrough) for d in vm.config.hardware.device):
                    raise RuntimeError("This test requires no running passthrough VM")
        prefix = args.endpoint + "-" + args.label
        if args.action == "snapshot":
            print(json.dumps(snapshot(content, host, ssh, prefix)), flush=True)
        elif args.action in {"prepare-active", "disable-both"}:
            request(content, host, ssh, TARGETS, args.action == "prepare-active", prefix)
        else:
            for suffix, bdf in zip(["a", "b"], TARGETS):
                baseline = request(content, host, ssh, TARGETS, True, prefix + "-prepare-" + suffix)
                if baseline["callOutcome"] != "success" or not all(p["enabled"] and p["active"] for p in baseline["after"]["pci"]):
                    raise RuntimeError("Both-active baseline not established")
                request(content, host, ssh, [bdf], False, prefix + "-disable-" + suffix)
            for log in ["hostd", "vmkernel"]:
                r = ssh.run("tail -n 1800 /var/run/log/" + log + ".log", check=False)
                (OUT / (prefix + "-" + log + ".log")).write_text(r.stdout)
    finally:
        Disconnect(si)


if __name__ == "__main__":
    main()
