import hashlib,json,re,urllib.parse,urllib.request
from pathlib import Path
from datetime import datetime,timezone
from fetch_iso_modules import Links
directory='https://vmware.lenovo.com/content/2026_09/Lenovo_Custom_ISO/8.0u3/'
filename='VMware-ESXi-8.0.3k-25595708-LNV-S15-20260825.zip'
out=Path(r'E:\Code\PCI\_artifacts\vmware-directpath-runs\20261113-esxi8-u3k-alias-validation-v1\input');out.mkdir(parents=True,exist_ok=True)

    for line in text.splitlines():
        if filename in line:

            if match:expected=match.group().lower();checksum_url=candidate

if start:
    with partial.open('rb') as f:
        for data in iter(lambda:f.read(1024*1024),b''):sha1.update(data);sha256.update(data)
with opener.open(urllib.request.Request(url,headers=headers),timeout=60) as r:
    assert r.status==(206 if start else 200),'Resume response mismatch'
    expected_total=start+int(r.headers['Content-Length']);count=start;next_report=count+64*1024*1024
    with partial.open('ab' if start else 'wb') as f:
        while True:
            data=r.read(1024*1024)
            if not data:break
            f.write(data);sha1.update(data);sha256.update(data);count+=len(data)
            if count>=next_report:print(json.dumps({'downloadedMiB':count//1048576,'totalMiB':expected_total//1048576}),flush=True);next_report=count+64*1024*1024

partial.rename(target)
receipt={'at':datetime.now(timezone.utc).isoformat(),'url':url,'checksumUrl':checksum_url,'filename':filename,'bytes':count,'sha1':sha1.hexdigest(),'vendorSha1Matched':True,'sha256':sha256.hexdigest()}
(out/'download-receipt.json').write_bytes((json.dumps(receipt,indent=2)+'\n').encode());print(json.dumps(receipt))
